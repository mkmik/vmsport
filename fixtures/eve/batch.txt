# EVE without a screen (EDIT/TPU/NODISPLAY): name, the EDIT command, its
# initialization file's lines (" | " between them, "-" for none), the
# files to TYPE afterwards. Every case starts from files/ as version 1.

init_end	EDIT/TPU/NODISPLAY/INITIALIZATION=I.EVE T.TXT	bottom | line 2 | uppercase word | move right	T.TXT;0

ambiguous	EDIT/TPU/NODISPLAY/INITIALIZATION=I.EVE T.TXT	e | bottom	T.TXT;0

no_init	EDIT/TPU/NODISPLAY/INITIALIZATION=NOSUCH.EVE T.TXT	-	T.TXT;*
